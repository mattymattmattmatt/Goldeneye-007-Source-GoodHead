# ===========================================================================
# GESVR - HEAD-AIM VR MODE
# ---------------------------------------------------------------------------
# Configured to play the game properly in VR without motion weapons: full
# stereo, head tracking, head aim, floating HUD, gun in its normal Source
# position. This is the Luke Ross style of VR conversion, and it avoids the
# entire motion-weapon problem area (skewing, snap-to-head, offset muzzle
# flash) which is not close to production quality.
#
# To go back to motion weapons later, set:
#   AimMode=gun / MotionControls=true / WeaponSetupHook=true
#   ViewmodelRenderablePatch=true / ShowWristHUD=true
#
# This whole file hot-reloads: save it while the game runs and changes apply in
# about a second.
# ===========================================================================

# ---------------------------------------------------------------------------
# How the stereo image reaches the headset.
#   compositor : real per-eye submit with the correct asymmetric frustum crop.
#                This is the one that should feel like you are IN the level.
#   sbs        : the old head-locked side-by-side quad. Stereo, but it reads as
#                a giant 3D television. Kept as a fallback.
#   both       : compositor plus the quad, for an A/B in one session.
# This file is hot-reloaded -- save it while the game runs and it takes effect.
DisplayMode=compositor

# Crops each eye's true asymmetric frustum out of the shared wide render.
# Turning this off reproduces the old too-big-world look; it is the single
# biggest contributor to correct VR scale.
UseEyeFrustumCrop=true

# The horizontal half of that crop is unambiguous. The vertical half depends on
# a sign convention that can only be settled in a headset: if the world sits too
# high or too low, set this false and only the horizontal crop applies.
UseVerticalCrop=true

# Multiplier on the WINDOW size for the eye render targets (only used when
# EyeRenderTargets is on). Sizing them to the HMD's recommended 2496x2688 froze
# the game and left only a sliver rendering -- that is the landmine recorded in
# HANDOFF.md, and it also breaks the 2D path, since the menu and HUD still draw
# to the 1280x720 backbuffer that the overlay capture reads.
# 1.5 gives 1920x1080 from a 1280x720 window: sharper, nowhere near the cliff.
# Try EyeRenderTargets=true with this at 1.25 or 1.5 before going higher.
EyeRenderScale=1.5

# Floating menu panel size and distance, in metres. The in-game character/level
# menu was too big and too close; raise the distance and/or drop the width until
# it sits comfortably. Both hot-reload.
MenuWidthMeters=2.4
MenuDistanceMeters=1.6

# Render each eye into its own engine render target instead of drawing both into
# the backbuffer one on top of the other.
#
# The backbuffer approach is what produces "a stereoscopic view with another
# wonky frame drawn on top" -- the second eye literally paints over the first in
# the same buffer. It also renders at window size (1280x720) rather than HMD
# size, which is where the softness comes from.
#
# This is what the l4d2vr reference does, and its IMatRenderContext vtable is
# byte-identical to ours, so the call is as safe here as there.
#
# Leave false for the first run of this build (so the menu-freeze fix is tested
# on its own). Once the menu is stable, set true and the double-image should go.
EyeRenderTargets=false

# IVModelRender vtable index of DrawModelExecute -- the hook that lets the mod
# move the weapon onto your hand. This was NEVER hooked before: offsets.h has an
# L4D2 byte signature that GE:S's engine does not match, which is why the gun
# could only ever follow your view angles. The index was measured in game with a
# per-slot call counter: [18]=128689 and [19]=128291 were the two hottest and
# differ by 398 -- Source's DrawModelSetup/DrawModelExecute pair.
ModelDrawExecuteSlot=19

# The other half of the pair. The weapon must be MOVED here, not at Execute:
# the log showed the viewmodel arriving at Execute with its origin exactly equal
# to the view origin, and changing it there does nothing because the bone
# matrices are already built. Setup runs first and takes the info by non-const
# reference, so a change lands in both the bones and the draw.
ModelDrawSetupSlot=18

# OFF - hooking slot 18 with a guessed DrawModelSetup signature crashed on the
# first stereo pass. The argument list does not match, so calling through
# corrupts the stack immediately. Do not enable until SetupProbe confirms the
# real signature.
WeaponSetupHook=false

# Redirects the viewmodel's GetRenderOrigin/GetRenderAngles to the controller
# pose. Writing ModelRenderInfo_t::origin provably does NOT move the gun (the
# write fires with a 27-unit displacement and nothing happens) because the root
# transform is built by SetupBones from these two accessors instead.
# Set false if the weapon renders wrong or the game becomes unstable.
ViewmodelRenderablePatch=false

# true  = the hand pose is applied only during the weapon's draw call. Cleaner,
#         but the gun snaps back to your head on any draw path we do not
#         intercept -- which is what happens while firing.
# false = the pose stays applied permanently. The gun stays on your hand through
#         firing, at the cost of the engine also using it for culling and
#         attachments, which is what made it warp.
#
# NEITHER IS CORRECT. Both are symptoms of a render-only override: the model is
# drawn at your hand while the weapon ENTITY still sits at your head, so the mesh
# ends up spanning two transforms (the stretching) and effects spawn from the old
# position (the offset muzzle flash). Try both and use whichever bothers you
# less; the real fix is to move the entity, which is the next piece of work.
ViewmodelScopedPose=true

# Safe naked capture of slot 18's actual arguments. Costs nothing; turn off once
# the signature is known.
SetupProbe=false

# Motion trace. Samples the whole hand -> weapon chain 4x/second while in a map.
# Run the calibration sequence in HANDOFF.md and send the log; each stage can
# then be verified independently instead of guessed at.
MotionDebug=false

# Re-run the slot counter (conflicts with the hook above; diagnosis only).
VtableProbe=false
# ^^^ TURN THIS ON FOR RESOLUTION ^^^
# Confirmed from the 21:26 log: "stereo pass #0 eyeRT=0 size=1280x720". With
# this off, each eye is captured from the 1280x720 window backbuffer and then
# upscaled to a ~2496x2688 headset eye -- that IS the blurriness. With it on,
# each eye renders into its own render target at true HMD resolution.
# This file hot-reloads: you can flip it while standing in a map and see the
# difference immediately. If anything renders wrong, flip it back.

# Only used when DisplayMode=sbs or both.
SbsWidthMeters=3.17
SbsDistance=1.0

# Menu pointer. Win32 posts real mouse messages to the game window and is the
# reliable path. The VGUI-internal path calls into a vgui2 vtable whose 2007
# layout is unconfirmed -- leave it off unless you are testing it.
# Aim menus with your HEAD (default) or the controller.
# In a map SteamVR sends almost no laser events to our overlay - measured at 2,
# versus hundreds at the main menu - because we are submitting stereo frames, so
# it gives the controller to the game instead. Head aiming works in both places:
# look at the item, pull the trigger. Set 'controller' to use the laser instead.
# auto (default) = controller pointer OUT of a map, head pointer IN one.
# That matches how SteamVR actually behaves: at the server-select menu the
# controller laser works well (hundreds of overlay events), but once we start
# submitting stereo frames SteamVR routes the controller to the game and sends
# almost none (measured: 2), so the head ray takes over there.
# Force with 'head' or 'controller'.
MenuAimSource=auto

# If the in-game character menu hangs the game again, set this false. It stops
# the head pointer from touching the OS cursor, so you lose the in-map cursor
# but keep a stable game.
MenuDriveCursor=true

# Draw our own crosshair marker into the menu image. MUST stay on: Source uses a
# HARDWARE cursor, composited by Windows and never present in the D3D backbuffer
# that we capture, so without this there is no cursor visible in the headset at
# all. The 'two cursors' were this marker plus the real Windows cursor on the
# desktop monitor -- only one of them is ever visible in VR.
DrawMenuCursor=true

# Aiming reticle drawn straight into each eye. The game's own crosshair never
# reaches the headset -- +crosshair 1 is set, but the 2D HUD is not part of what
# we capture. In head-aim mode you aim with the centre of your view, so a centre
# reticle is both correct and independent of the game's HUD.
VRReticle=true

MenuInputWin32=true
MenuInputVguiInternal=false

# Kill switch for ALL OS-level cursor driving (SetCursorPos, SetForegroundWindow,
# PostMessage). Set false to rule the whole menu-input subsystem out at once.
# (duplicate of the setting above, removed) MenuDriveCursor=true

# SetCursorPos moves the real Windows cursor. It is a system-wide call that can
# transact with the desktop input thread; PostMessage cannot block. If the game
# freezes shortly after the menu appears, set this false -- Source's inputsystem
# consumes WM_MOUSEMOVE on its own, so the laser may still work without it.
MenuUseSetCursorPos=true

# Keep the game window on a single monitor. At higher render resolutions the
# window was reported spanning both displays. Only the captured backbuffer
# reaches the headset, so a stretched desktop window costs nothing visually, but
# it is disruptive to work around. Handled on the input worker thread, which
# owns all USER32 - doing window work from the render thread is what deadlocked
# the game earlier in this project.
KeepWindowOnPrimaryMonitor=false

# How a menu click is delivered. The freeze happens immediately after a click is
# posted, every time. PostMessage synthesises a message with no hardware input
# behind it; VGUI calls SetMouseCapture on button-down, and a synthetic press
# whose physical button was never down can leave it waiting for a release that
# cannot arrive. SendInput goes through the real input queue like a physical
# mouse. If the game freezes the moment you click, try true.
MenuClickViaSendInput=false

# How often the menu talks to the SteamVR compositor, in milliseconds.
# This is NOT just a keepalive. OpenVR's Vulkan Submit drains the graphics
# queue, so its cost is proportional to how much game rendering piled up since
# the last submit. At the old 500ms spacing the engine free-ran at ~225fps,
# leaving ~110 frames to drain -- an 83-150ms freeze every half second, which
# is the stutter you were feeling. 33ms cuts that to about 5ms.
#   33  = default, smooth
#   11  = display cadence, smoothest; try it if 33 is stable
#   500 = old behaviour, for comparison
MenuKeepaliveMs=33


# SteamVR's desktop mirror window. It competes with the game for Win32 focus,
# and Source throttles rendering hard (and mutes audio) whenever its own window
# is not the active app. A VR mod gains nothing from it. Leave off.
ShowMirrorWindow=false

# How often to re-suppress the SteamVR Theater overlay, in milliseconds.
# Each pass is ~10 IPC round-trips to vrserver. 0 = every frame (old behaviour).
TheaterHideThrottleMs=2000

EnableNetVR=true
# head = look and shoot with the headset
# gun  = weapons follow the right controller (default)
AimMode=head
MotionControls=false

# Gun grip correction, degrees. A controller's forward axis is not a barrel
# axis; this rotates the weapon down into a natural grip. 45 is what L4D2VR and
# HaloCEVR use. If the gun points too high, raise it; too low, lower it.
# Set 0 to aim straight down the controller's own axis.
# Confirmed in play: at 0, pointing straight ahead shoots ~45 degrees HIGH.
# Positive pitch is DOWN in Source, so 45 brings the shot onto your aim.
# Nudge in 5-degree steps if it is slightly off; this hot-reloads.
# Viewmodel FOV. 0 = use the world FOV, which for the HMD's superset frustum is
# ~106 degrees -- very wide, and it drags the weapon toward the centre of your
# view and makes it disagree with the world-space muzzle effect.
# Try 54-75 to put the gun back where Source normally draws it. Hot-reloads, so
# nudge it while looking at the gun.
ViewmodelFov=0

GunGripAngle=45.0

# Per-weapon viewmodel poses from the table in weapons.cpp (pp7, dd44, klobb,
# ar33, rcp90, sniper, shotguns, knife, grenades...). That table was completely
# unreachable until now -- its only caller had no call sites -- so every weapon
# used one generic pose.
PerWeaponOffsets=false

# Two-handed grip: bring the off hand up near the barrel and the gun aims along
# the line between your hands. Skipped for melee, dual-wieldables and throwables.
TwoHandedGrip=true

# HOLD the off-hand grip button to go two-handed, HaloCEVR-style. Previously it
# engaged purely on hand distance, which fired whenever your hands drifted near
# each other. Set false to go back to distance-only.
TwoHandedNeedsGrip=true

# Fine gun placement relative to your hand, in Source units (1 unit ~= 1 inch),
# as forward,right,up. Positive forward pulls the weapon back toward you.
# --- Weapon alignment -------------------------------------------------------
# Hand tracking is live, so these two dial the gun into your hand. Both
# hot-reload: edit, save, and the change lands in about a second while you are

# Position, in Source units (~1 inch), as forward,right,up relative to your

ViewmodelOffset=0,0,0

# Rotation of the WEAPON MODEL only, as pitch,yaw,roll in degrees.
